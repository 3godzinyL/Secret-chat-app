cmake_minimum_required(VERSION 3.21)
project(umbra_native_crypto LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
if(NOT DEFINED NODE_INCLUDE_DIR)
  message(FATAL_ERROR "Pass -DNODE_INCLUDE_DIR=path/to/node/include/node")
endif()
add_library(umbra_crypto MODULE umbra_crypto.cc)
target_include_directories(umbra_crypto PRIVATE "${NODE_INCLUDE_DIR}")
target_link_libraries(umbra_crypto PRIVATE OpenSSL::Crypto)
set_target_properties(umbra_crypto PROPERTIES PREFIX "" SUFFIX ".node")
if(WIN32)
  # Windows Node-API modules resolve Node symbols from node.lib.
  if(NOT DEFINED NODE_IMPORT_LIBRARY)
    message(FATAL_ERROR "Pass -DNODE_IMPORT_LIBRARY=path/to/node.lib")
  endif()
  target_link_libraries(umbra_crypto PRIVATE "${NODE_IMPORT_LIBRARY}")
endif()
